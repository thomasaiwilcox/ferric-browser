cmake_minimum_required(VERSION 3.20)
project(ferric_evidence_tests LANGUAGES CXX)
set(CMAKE_AUTOMOC ON)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 REQUIRED COMPONENTS Test Qml WebEngineCore WebEngineQuick)
add_executable(evidence_tests tst_evidence.cpp ../../src/request_interceptor.cpp ../../src/request_interceptor.h)
target_include_directories(evidence_tests PRIVATE ../../src)
target_link_libraries(evidence_tests PRIVATE Qt6::Test Qt6::Qml Qt6::WebEngineCore Qt6::WebEngineQuick)
enable_testing()
add_test(NAME evidence COMMAND evidence_tests)
