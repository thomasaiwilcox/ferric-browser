cmake_minimum_required(VERSION 3.20)
project(ferric_native_insert LANGUAGES CXX)
set(CMAKE_AUTOMOC ON)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 REQUIRED COMPONENTS Quick WebEngineQuick DBus Test)
add_executable(native_insert tst_native_insert.cpp native_probe_input.cpp ../../src/browser_key_router.cpp ../../src/browser_page_pointer_adapter.cpp ../../src/hint_frame_broker.cpp)
target_link_libraries(native_insert PRIVATE Qt6::Quick Qt6::WebEngineQuick Qt6::DBus Qt6::Test)
find_package(PkgConfig REQUIRED)
pkg_check_modules(NATIVE_PROBE_XTEST IMPORTED_TARGET x11 xtst)
if(NATIVE_PROBE_XTEST_FOUND)
    target_link_libraries(native_insert PRIVATE PkgConfig::NATIVE_PROBE_XTEST)
    target_compile_definitions(native_insert PRIVATE FERRIC_NATIVE_PROBE_XTEST=1)
endif()
option(FERRIC_NATIVE_PROBE_ASAN "Instrument native adapter ownership probes" OFF)
if(FERRIC_NATIVE_PROBE_ASAN)
    target_compile_options(native_insert PRIVATE -fsanitize=address -fno-omit-frame-pointer -g)
    target_link_options(native_insert PRIVATE -fsanitize=address)
endif()
target_compile_definitions(native_insert PRIVATE
    FERRIC_NATIVE_PROBE=1
    FERRIC_SCRIPT_PATH="${CMAKE_CURRENT_SOURCE_DIR}/../../qml/scripts/BrowserScripts.js"
    FERRIC_POLICY_FIXTURE_PATH="${CMAKE_CURRENT_SOURCE_DIR}/../fixtures/site-policy/interaction.html"
    FERRIC_DRAG_FIXTURE_PATH="${CMAKE_CURRENT_SOURCE_DIR}/../fixtures/drag-probe.html")
enable_testing()
add_test(NAME native_insert COMMAND native_insert)
set_tests_properties(native_insert PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software;QTWEBENGINE_CHROMIUM_FLAGS=--disable-gpu")
add_subdirectory(../native-input-lifetime native-input-lifetime)
